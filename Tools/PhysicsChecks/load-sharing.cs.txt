// Unity MCP execute_code method body. Run in the prepared __PocketPhysicsReview scene.
// The old snapshot is rounded to intended cells: this is a layout reconstruction, not input replay.
if (!Application.isPlaying || UnityEngine.SceneManagement.SceneManager.GetActiveScene().name != "__PocketPhysicsReview"
    || GameManager.Instance != null || BlockController.AllBlocks.Count != 0)
    throw new Exception("Use an empty __PocketPhysicsReview play-mode scene with no game or blocks.");
var bf = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic;
var sf = System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.NonPublic;
object Call(BlockController b, string name, params object[] args) => typeof(BlockController).GetMethod(name, bf).Invoke(b, args);
Vector2 V(Newtonsoft.Json.Linq.JToken t) => new Vector2((float)t[0], (float)t[1]);
var checks = new List<object>(); var errors = new List<string>(); var timings = new List<object>();
void Check(string name, bool passed, object details = null) => checks.Add(new { name, passed, details });
Application.LogCallback captureError = (m, s, t) => { if (t == LogType.Error || t == LogType.Exception || t == LogType.Assert) errors.Add(m); };
Application.logMessageReceived += captureError;
var previousSimulation = Physics2D.simulationMode; float previousTimeScale = Time.timeScale;
var data = Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText("Tools/PhysicsChecks/Fixtures/tilted-branch.json"));
GameObject root = null;
var pieces = new List<BlockController>();
void Start()
{
    if (root != null) UnityEngine.Object.DestroyImmediate(root);
    root = new GameObject("Load sharing review"); pieces.Clear();
}
void Terrain(bool mirrored, float offset)
{
    foreach (var entry in data["terrain"])
    {
        var go = new GameObject("Captured terrain"); go.transform.SetParent(root.transform);
        Vector2 p = V(entry["position"]); p.x = (mirrored ? -p.x : p.x) + offset;
        go.transform.position = p; var box = go.AddComponent<BoxCollider2D>();
        box.size = V(entry["size"]); box.edgeRadius = (float)entry["radius"];
    }
    Physics2D.SyncTransforms();
}
BlockController Create(string shape, Vector2 position, float angle, string variant = "Normal", bool initialize = true)
{
    var prefab = UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/Blocks/Block_" + shape + ".prefab");
    var go = UnityEngine.Object.Instantiate(prefab, position, Quaternion.Euler(0, 0, angle), root.transform);
    var block = go.GetComponent<BlockController>(); block.enabled = false;
    block.ApplyData(UnityEditor.AssetDatabase.LoadAssetAtPath<BlockData>("Assets/Data/Blocks/" + variant + ".asset"));
    if (initialize) Call(block, "InitDynamicControlBody"); pieces.Add(block); return block;
}
BlockController Place(Newtonsoft.Json.Linq.JToken entry, bool mirrored, float offset)
{
    string shape = ((string)entry["name"]).Replace("Block_", "").Replace("(Clone)", "");
    float angle = Mathf.Round((float)entry["angle"] / 90) * 90;
    Vector2 p = V(entry["position"]); p = new Vector2(Mathf.Round(p.x), Mathf.Round(p.y));
    if (mirrored)
    {
        p.x = -p.x; angle = -angle;
        if (shape == "L") shape = "J"; else if (shape == "J") shape = "L";
        else if (shape == "S") shape = "Z"; else if (shape == "Z") shape = "S";
        // I and O prefab origins are offset from their mirror axes by half a cell.
        if (shape == "I" || shape == "O") p += (Vector2)(Quaternion.Euler(0, 0, angle) * Vector3.right);
    }
    p.x += offset;
    var block = Create(shape, p, angle); Call(block, "BeginPhysicsLanding"); return block;
}
void Step(int frames, bool reverse = false)
{
    for (int i = 0; i < frames; i++)
    {
        for (int j = 0; j < pieces.Count; j++)
        {
            var block = pieces[reverse ? pieces.Count - 1 - j : j];
            if (block != null && block.HasLanded) Call(block, "HandleLandedMaintenance");
        }
        Physics2D.Simulate(.02f);
    }
}
bool Exact(BlockController b)
{
    if (b == null || !b.IsGridStable) return false;
    var cells = new List<Vector2>(); b.GetWorldCellCenters(cells);
    return cells.All(c => Mathf.Abs(c.x - Mathf.Round(c.x)) < .0001f && Mathf.Abs(c.y - Mathf.Round(c.y)) < .0001f)
        && Mathf.Abs(Mathf.DeltaAngle(b.transform.eulerAngles.z, Mathf.Round(b.transform.eulerAngles.z / 90) * 90)) < .0001f;
}
List<BlockController> Component(BlockController seed) => (List<BlockController>)typeof(BlockController)
    .GetMethod("CollectGridStableComponent", sf).Invoke(null, new object[] { seed, null });
bool Balanced(List<BlockController> component) => (bool)typeof(BlockController)
    .GetMethod("CanBalanceGridStructure", sf).Invoke(null, new object[] { component, null });
BlockController Drop(string shape, float left, float bottom, float angle = 0)
{
    var b = Create(shape, Vector2.zero, angle, initialize: false);
    var cells = new List<Vector2>(); b.GetWorldCellCenters(cells);
    Call(b, "SetPosition", new Vector3(left - cells.Min(c => c.x), bottom + 3 - cells.Min(c => c.y), 0));
    Call(b, "ResetControlTargets"); Call(b, "InitDynamicControlBody");
    for (int i = 0; i < 2000 && !b.HasLanded; i++) Call(b, "SteerWhileFalling");
    if (!b.HasLanded) throw new Exception("Drop did not land: left=" + left + " bottom=" + bottom + " angle=" + angle + " actual=" + b.transform.position + " count=" + pieces.Count);
    return b;
}
try
{
    Time.timeScale = 0; Physics2D.simulationMode = SimulationMode2D.Script;
    foreach (bool mirrored in new[] { false, true })
    foreach (float offset in new[] { 0f, 200f })
    {
        Start(); Terrain(mirrored, offset); string label = "Reconstructed tower mirror=" + mirrored + " offset=" + offset;
        int firstRelease = -1; bool greedyRejects = false;
        for (int i = 0; i < data["blocks"].Count(); i++)
        {
            Place(data["blocks"][i], mirrored, offset);
            if (firstRelease < 0 && pieces.Any(b => !Exact(b))) firstRelease = i;
            if (i == 33 && pieces[10].IsGridStable)
                greedyRejects = typeof(BlockController).GetMethod("FindUnstableGridBlockByLoadPath", sf)
                    .Invoke(null, new object[] { Component(pieces[10]), null }) != null;
            Step(25, mirrored); // real physics and production settling between every placement
        }
        Check(label + " keeps all 50 placements exact", firstRelease < 0 && pieces.All(Exact), new { firstRelease });
        Check(label + " exercises the rejected greedy load path", greedyRejects);
        var positions = pieces.Select(b => b.transform.position).ToArray(); Step(6000, mirrored);
        float drift = pieces.Select((b, i) => Vector3.Distance(positions[i], b.transform.position)).Max();
        Check(label + " stays exact for 120 seconds", pieces.All(Exact) && drift < .00001f, new { drift });
        var component = Component(pieces[10]);
        Check(label + " admits balanced reactions in reverse node order", Balanced(component.AsEnumerable().Reverse().ToList()));
        // Translation fixtures test equilibrium numerics, not the scene's world-centered control bounds.
        if (offset != 0) continue;
        // Control-path landings after the reconstruction. Alternate four rotations on new
        // full-width supports on the left foundation, while the outer branch stays load-bearing.
        float left = mirrored ? offset + 5 : offset - 6;
        var allCells = new List<Vector2>(); foreach (var b in pieces) { var c = new List<Vector2>(); b.GetWorldCellCenters(c); allCells.AddRange(c); }
        float top = allCells.Max(c => c.y) + 1;
        var first = Drop("O", left, top);
        float firstY = first.GetComponent<Rigidbody2D>().position.y;
        int addedFailure = first.IsGridStable ? -1 : 0;
        for (int i = 1; i < 30; i++) { Drop("O", left, firstY + 2 * i, (i % 4) * 90); if (addedFailure < 0 && pieces.Any(b => !Exact(b))) addedFailure = i; Step(25); }
        Check(label + " remains exact through 30 further cast-driven landings", pieces.All(Exact), new {addedFailure, blocks=pieces.Select((b,i)=>new {i,b.IsGridStable,position=b.transform.position.ToString("F6"),rotation=b.transform.eulerAngles.z}).Where((b,i)=>!Exact(pieces[i])).ToArray()});
        var watch = System.Diagnostics.Stopwatch.StartNew(); bool balanced = Balanced(Component(first)); watch.Stop();
        timings.Add(new { label, count = pieces.Count, balanced, milliseconds = watch.Elapsed.TotalMilliseconds });
        Step(1500);
        Check(label + " tall stack stays exact after a further 30 seconds", pieces.All(Exact));
    }
    // The same supported branch must lose ownership under a genuine outboard overload.
    foreach (bool mirrored in new[] { false, true })
    {
        Start(); Terrain(mirrored, 0);
        for (int i = 0; i <= 33; i++) Place(data["blocks"][i], mirrored, 0);
        var outer = pieces[33]; var rb = outer.GetComponent<Rigidbody2D>(); rb.mass = 100;
        Call(outer, "ValidateGridStructureAfterLanding");
        Check("Outboard overload releases, mirror=" + mirrored, !outer.IsGridStable && !pieces[10].IsGridStable);
        Vector2 before = rb.position; Step(500);
        Check("Outboard overload physically moves, mirror=" + mirrored, Vector2.Distance(before, rb.position) > .05f);
    }
    Start(); Terrain(false, 0);
    for (int i = 0; i < 50; i++) Place(data["blocks"][i], false, 0);
    pieces[10].ApplyJolt(new Vector2(2, 0));
    Check("Equilibrium does not intercept deliberate jolts", pieces.All(b => !b.IsGridStable));
    Step(250); Check("Jolted structure never re-registers", pieces.All(b => !b.IsGridStable));
    Start(); Terrain(false, 0);
    for (int i = 0; i <= 33; i++) Place(data["blocks"][i], false, 0);
    // Remove every foundation collider: no internally balanced loop can supply gravity.
    foreach (var box in root.GetComponentsInChildren<BoxCollider2D>().Where(c => c.GetComponentInParent<BlockController>() == null)) box.enabled = false;
    Physics2D.SyncTransforms(); Call(pieces[33], "ValidateGridStructureAfterLanding");
    Check("Removing all terrain releases the structure", pieces.All(b => !b.IsGridStable));
    Step(150); Check("Unsupported structure falls", pieces.All(b => b.GetComponent<Rigidbody2D>().linearVelocity.y < -1));
}
finally
{
    if (root != null) UnityEngine.Object.DestroyImmediate(root);
    Physics2D.SyncTransforms(); Physics2D.simulationMode = previousSimulation; Time.timeScale = previousTimeScale;
    Application.logMessageReceived -= captureError;
}
System.IO.File.WriteAllText("Library/load-sharing-checks.json", Newtonsoft.Json.JsonConvert.SerializeObject(new { checks, errors, timings }, Newtonsoft.Json.Formatting.Indented));
int failed = Newtonsoft.Json.Linq.JArray.FromObject(checks).Count(c => !(bool)c["passed"]);
return new { passed = checks.Count - failed, failed, errors, timings, report = "Library/load-sharing-checks.json" };
